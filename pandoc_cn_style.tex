% Pandoc PDF style for Chinese survey reports (Windows 版本)
\usepackage{fontspec}
\usepackage{xeCJK}
\usepackage{indentfirst}
\usepackage{titlesec}
\usepackage{graphicx}
\usepackage{caption}
\usepackage{float}
\usepackage{etoolbox}

% Windows fonts configuration
\newCJKfontfamily\fontheiti{SimHei}
\newCJKfontfamily\fontkaiti{KaiTi}
\newCJKfontfamily\fontfangsong{FangSong}
\newCJKfontfamily\fontsongti{SimSun}

% 正文：宋体 12pt 1.5倍行距
\AtBeginDocument{\fontsize{12pt}{18pt}\selectfont}
\setlength{\parindent}{2em}
\setlength{\parskip}{0pt}
\linespread{1.5}

% 不自动生成标题
\AtBeginDocument{\let\maketitle\relax}

% 表格字体
\AtBeginEnvironment{tabular}{\fontsize{10.5pt}{15.75pt}\selectfont}
\AtBeginEnvironment{longtable}{\fontsize{10.5pt}{15.75pt}\selectfont}

% ==================== 标题格式 ====================
\titleformat{\section}
  {\centering\fontheiti\fontsize{16pt}{24pt}\selectfont\bfseries}
  {\thesection}{0.8em}{}
\titleformat{\subsection}
  {\fontheiti\fontsize{14pt}{21pt}\selectfont\bfseries}
  {\thesubsection}{0.6em}{}
\titleformat{\subsubsection}
  {\fontkaiti\fontsize{12pt}{18pt}\selectfont\bfseries}
  {\thesubsubsection}{0.6em}{}
\titleformat{\paragraph}[hang]
  {\fontsize{12pt}{18pt}\selectfont}
  {\theparagraph}{0.6em}{}

\titlespacing*{\section}{0pt}{24pt}{16pt}
\titlespacing*{\subsection}{0pt}{16pt}{12pt}
\titlespacing*{\subsubsection}{0pt}{12pt}{8pt}
\titlespacing*{\paragraph}{0pt}{8pt}{6pt}

% ==================== 页码居底 ====================
\pagestyle{plain}
\raggedbottom
\usepackage{fancyhdr}
\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\thepage}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}

% ==================== 图片设置 ====================
\floatplacement{figure}{htbp}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.8}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.8}
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}
\setlength{\intextsep}{10pt plus 2pt minus 2pt}
\setkeys{Gin}{width=0.9\linewidth,keepaspectratio}
\captionsetup[figure]{font={small},justification=centering,singlelinecheck=false,skip=6pt}
